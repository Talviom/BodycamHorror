# 0.9, пункт 4: бинт. Полоска здоровья на плёнке, «ПЕРЕВЯЗКА» с полоской хода, 3D-рулон и обматывание,
# бинт снимает страх (-20), отпущенная кнопка отменяет перевязку и бинт не тратится.
# Запуск: gradlew runClient -PbhTest=tools/tests/bandage.txt
# Результат: [bh-test] DONE failures=0, скриншоты run/screenshots/090_bandage_*.png

bh stop
bh bot remove all
wait 20
publish false adventure 25571
expect 25571
gamemode adventure @p
# Вне лобби: в лобби урон по игрокам выключен (LobbyEvents).
# Сначала телепорт и пауза: иначе fill пишет «That position is not loaded» и игрок падает в пустоту.
tp @p 2000 100 2000 0 10
wait 20
fill 1990 99 1990 2010 99 2010 minecraft:stone
fill 1990 100 2006 2010 103 2006 minecraft:stone_bricks
tp @p 2000 100 2000 0 10
time set 6000
# Без самолечения, иначе за время проверки здоровье само доходит до полного.
gamerule naturalRegeneration false
weather clear
effect clear @p
clear @p
item replace entity @p hotbar.8 with bodycamhorror:flashlight
item replace entity @p weapon.mainhand with bodycamhorror:bandage 3
wait 20
shot 090_bandage_hp_full

# --- урон: плёнка теряет кадры, потерянный кусок тлеет кровью ---
# Вне матча команда damage в тестовом мире не ранит (Target is invulnerable, даже out_of_world),
# поэтому «рана» здесь это большее максимальное здоровье: 20 из 50 = 40 %.
attribute @p minecraft:generic.max_health base set 50
data get entity @p Health
wait 6
shot 090_bandage_hp_hit
wait 40
shot 090_bandage_hp_low
execute as @p run bh horror fear 80
expect Fear set to 80

# --- перевязка целиком: 40 тиков ---
use on
wait 8
shot 090_bandage_wrap_start
wait 12
shot 090_bandage_wrap_mid
wait 14
shot 090_bandage_wrap_tug
wait 16
use off
wait 4
execute as @p run bh horror fear
expect Fear: 6
clear @p bodycamhorror:bandage 0
expect 2
wait 30
shot 090_bandage_healed

# --- отмена: отпустил раньше, бинт цел, страх тот же ---
execute as @p run bh horror fear 50
use on
wait 15
use off
wait 6
clear @p bodycamhorror:bandage 0
expect 2
execute as @p run bh horror fear
expect Fear: 50

# --- критическое здоровье: кадры темнеют до крови и мерцают ---
attribute @p minecraft:generic.max_health base set 120
wait 30
shot 090_bandage_hp_critical

# --- вид со спины: обе руки перед грудью, рулон ходит кругами ---
view third
use on
wait 16
shot 090_bandage_third
use off
view first

lang ru_ru
# Перезагрузка ресурсов после смены языка занимает несколько секунд.
wait 200
attribute @p minecraft:generic.max_health base set 20
wait 10
attribute @p minecraft:generic.max_health base set 60
wait 10
data get entity @p Health
clear @p bodycamhorror:bandage 0
item replace entity @p weapon.mainhand with bodycamhorror:bandage 2
wait 10
use on
wait 20
shot 090_bandage_wrap_ru
use off
lang en_us
wait 200
attribute @p minecraft:generic.max_health base set 20
